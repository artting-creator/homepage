type Audio = {
  /** 標題 */
  title: string;
  /** 音頻的網絡鏈接 */
  url: string;
};

type AudioWithOptionalTitle = {
  /** 標題 */
  title?: string;
  /** 音頻的網絡鏈接 */
  url: string;
};

/**
 * 播放給定的音頻; 如果該音頻沒在播放列表中, 則會加入到播放列表.
 *
 * @param type 背景音樂 ('bgm') 或音效 ('ambient')
 * @param audio 要播放的音頻; 如果音頻沒有設置標題 (`title`), 則會從鏈接 (`url`) 提取文件名作爲標題
 *
 * @example
 * // 將給定鏈接作爲背景音樂播放
 * playAudio('bgm', { url: 'http://commondatastorage.googleapis.com/codeskulptor-demos/DDR_assets/Kangaroo_MusiQue_-_The_Neverwritten_Role_Playing_Game.mp3' });
 *
 * @example
 * // 爲給定鏈接設置標題, 並作爲背景音樂播放
 * playAudio('bgm', { title: 'Kangaroo Music', url: 'http://commondatastorage.googleapis.com/codeskulptor-demos/DDR_assets/Kangaroo_MusiQue_-_The_Neverwritten_Role_Playing_Game.mp3' });
 */
declare function playAudio(type: 'bgm' | 'ambient', audio: AudioWithOptionalTitle): void;

/**
 * 暫停音樂
 *
 * @param type 背景音樂 ('bgm') 或音效 ('ambient')
 */
declare function pauseAudio(type: 'bgm' | 'ambient'): void;

/**
 * 獲取播放列表
 *
 * @param type 背景音樂 ('bgm') 或音效 ('ambient')
 * @returns 播放列表
 */
declare function getAudioList(type: 'bgm' | 'ambient'): Audio[];

/**
 * 完全替換播放列表爲 `audio_list`
 *
 * @param type 背景音樂 ('bgm') 或音效 ('ambient')
 * @param audio_list 新的播放列表; 如果其中音頻沒有設置標題 (`title`), 則會從鏈接 (`url`) 提取文件名作爲標題
 */
declare function replaceAudioList(type: 'bgm' | 'ambient', audio_list: AudioWithOptionalTitle[]): void;

/**
 * 向播放列表末尾添加不存在的音頻, 不會重複添加同 `title` 或 `url` 的音頻
 *
 * @param type 背景音樂 ('bgm') 或音效 ('ambient')
 * @param audio_list 要插入的音頻列表; 如果其中音頻沒有設置標題 (`title`), 則會從鏈接 (`url`) 提取文件名作爲標題
 */
declare function appendAudioList(type: 'bgm' | 'ambient', audio_list: AudioWithOptionalTitle[]): void;

type AudioSettings = {
  /** 是否啓用 */
  enabled: boolean;
  /**
   * 當前播放模式
   * - repeat_one: 單曲循環
   * - repeat_all: 全部循環
   * - shuffle: 隨機播放
   * - play_one_and_stop: 播放一首後停止
   */
  mode: 'repeat_one' | 'repeat_all' | 'shuffle' | 'play_one_and_stop';
  /** 是否靜音 */
  muted: boolean;
  /** 當前音量 (0-100) */
  volume: number;
};

/**
 * 獲取音頻設置
 *
 * @param type 背景音樂 ('bgm') 或音效 ('ambient')
 * @returns 音頻設置
 */
declare function getAudioSettings(type: 'bgm' | 'ambient'): AudioSettings;

/**
 * 修改音頻設置, 如果某字段不存在, 則使用原本的設置.
 *
 * @param type 背景音樂 ('bgm') 或音效 ('ambient')
 * @param settings 要修改的音頻設置
 *
 * @example
 * // 將背景音樂設置爲單曲循環
 * setAudioSettings('bgm', { mode: 'repeat_one' });
 *
 * @example
 * // 將音效設置爲靜音
 * setAudioSettings('ambient', { muted: true });
 *
 * @example
 * // 將背景音樂音量設置爲 50%
 * setAudioSettings('bgm', { volume: 50 });
 */
declare function setAudioSettings(type: 'bgm' | 'ambient', settings: Partial<AudioSettings>): void;

type ReplaceCharacterOptions = {
  /** 酒館網頁應該防抖渲染 (debounced)、立即渲染 (immediate) 還是不刷新前端顯示 (none)? 默認爲性能更好的防抖渲染 */
  render?: 'debounced' | 'immediate' | 'none';
};

/**
 * 完全替換 `character_name` 角色卡的內容爲 `character`
 *
 * @param character_name 角色卡名稱
 * @param character 角色卡數據
 * @param options 可選選項
 *   - `render:'debounced'|'immediate'|'none'`: 酒館網頁應該防抖渲染 (debounced)、立即渲染 (immediate) 還是不刷新前端顯示 (none)? 默認爲性能更好的防抖渲染
 *
 * @throws 如果角色卡不存在, 將會拋出異常
 * @throws 如果訪問後端失敗, 將會拋出異常
 *
 * @example
 * // 爲角色卡更改開場白
 * const character = await getCharacter('角色卡名稱');
 * character.first_messages = ['新的開場白1', '新的開場白2'];
 * await replaceCharacter('角色卡名稱', character);
 *
 * @example
 * // 清空角色卡的局部正則
 * const character = await getCharacter('角色卡名稱');
 * character.extensions.regex_scripts = [];
 * await replaceCharacter('角色卡名稱', character);
 *
 * @example
 * // 更換角色卡頭像
 * const character = await getCharacter('角色卡名稱');
 * character.avatar = await fetch('https://example.com/avatar.png').then(response => response.blob());
 * await replaceCharacter('角色卡名稱', character);
 */
declare function replaceCharacter(
  character_name: Exclude<string, 'current'>,
  character: TypeFest.PartialDeep<Character>,
  options?: ReplaceCharacterOptions,
): Promise<void>;

type CharacterUpdater = ((character: Character) => Character) | ((character: Character) => Promise<Character>);

/**
 * 用 `updater` 函數更新 `character_name` 角色卡
 *
 * @param character_name 角色卡名稱
 * @param updater 用於更新角色卡的函數. 它應該接收角色卡內容作爲參數, 並返回更新後的角色卡內容.
 * @param options 可選選項
 *   - `render:'debounced'|'immediate'|'none'`: 如果對角色卡進行操作, 應該防抖渲染 (debounced)、立即渲染 (immediate) 還是不刷新前端顯示 (none)? 默認爲性能更好的防抖渲染
 *
 * @returns 更新後的角色卡內容
 *
 * @throws 如果角色卡不存在, 將會拋出異常
 * @throws 如果訪問後端失敗, 將會拋出異常
 *
 * @example
 * // 爲角色卡添加一個開場白
 * await updateCharacterWith('角色卡名稱', character => {
 *   character.first_messages.push('新的開場白');
 *   return character;
 * });
 *
 * @example
 * // 清空角色卡的局部正則
 * await updateCharacterWith('角色卡名稱', character => {
 *   character.extensions.regex_scripts = [];
 *   return character;
 * });
 *
 * @example
 * // 更換角色卡頭像
 * await updateCharacterWith('角色卡名稱', async character => {
 *   character.avatar = await fetch('https://example.com/avatar.png').then(response => response.blob());
 *   return character;
 * });
 */
declare function updateCharacterWith(
  character_name: TypeFest.LiteralUnion<'current', string>,
  updater: CharacterUpdater,
): Promise<Character>;

type ChatMessageSwiped = {
  message_id: number;
  name: string;
  role: 'system' | 'assistant' | 'user';
  is_hidden: boolean;
  swipe_id: number;
  swipes: string[];
  swipes_data: Record<string, any>[];
  swipes_info: Record<string, any>[];

type GetChatMessagesOption = {
  /** 按 role 篩選消息; 默認爲 `'all'` */
  role?: 'all' | 'system' | 'assistant' | 'user';
  /** 按是否被隱藏篩選消息; 默認爲 `'all'` */
  hide_state?: 'all' | 'hidden' | 'unhidden';
  /** 是否包含未被 AI 使用的消息頁信息, 如沒選擇的開局、通過點擊箭頭重 roll 的樓層. 如果不包含則返回類型爲 `ChatMessage`, 否則返回類型爲 `ChatMessageSwiped`; 默認爲 `false` */
  include_swipes?: boolean;
};

/**
 * 獲取聊天消息, 僅獲取每樓被 AI 使用的消息頁
 *
 * @param range 要獲取的消息樓層號或樓層範圍, 如 `0`, `'0-{{lastMessageId}}'`, `-1` 等. 負數表示深度, 如 `-1` 表示最新的消息樓層, `-2` 表示倒數第二條消息樓層.
 * @param option 可選選項
 *   - `role:'all'|'system'|'assistant'|'user'`: 按 role 篩選消息; 默認爲 `'all'`
 *   - `hide_state:'all'|'hidden'|'unhidden'`: 按是否被隱藏篩選消息; 默認爲 `'all'`
 *   - `include_swipes:false`: 不包含未被 AI 使用的消息頁信息
 *
 * @returns 一個 `ChatMessage` 數組, 包含指定樓層範圍內實際存在的所有樓層, 依據 message_id 從低到高排序; 如果範圍內完全不存在樓層 (如目前只有 3 樓, 但範圍爲 `4-5`), 則返回空數組
 *
 * @example
 * // 僅獲取第 10 樓被 AI 使用的消息頁
 * const chat_messages = getChatMessages(10);
 * const chat_messages = getChatMessages('10');
 * const chat_messages = getChatMessages('10', { include_swipes: false });
 *
 * @example
 * // 獲取最新樓層被 AI 使用的消息頁
 * const chat_message = getChatMessages(-1)[0];  // 或 getChatMessages('{{lastMessageId}}')[0]
 *
 * @example
 * // 獲取所有樓層被 AI 使用的消息頁
 * const chat_messages = getChatMessages('0-{{lastMessageId}}');
 */
declare function getChatMessages(
  range: string | number,
  { role, hide_state, include_swipes }?: Omit<GetChatMessagesOption, 'include_swipes'> & { include_swipes?: false },
): ChatMessage[];

/**
 * 獲取聊天消息, 獲取每樓所有的消息頁, 包含未被 AI 使用的消息頁消息
 *
 * @param range 要獲取的消息樓層號或樓層範圍, 如 `0`, `'0-{{lastMessageId}}'`, `-1` 等. 負數表示深度, 如 `-1` 表示最新的消息樓層, `-2` 表示倒數第二條消息樓層.
 * @param option 可選選項
 *   - `role:'all'|'system'|'assistant'|'user'`: 按 role 篩選消息; 默認爲 `'all'`
 *   - `hide_state:'all'|'hidden'|'unhidden'`: 按是否被隱藏篩選消息; 默認爲 `'all'`
 *   - `include_swipes:true`: 包含未被 AI 使用的消息頁信息
 *
 * @returns 一個 `ChatMessage` 數組, 包含指定樓層範圍內實際存在的所有樓層, 依據 message_id 從低到高排序; 如果範圍內完全不存在樓層 (如目前只有 3 樓, 但範圍爲 `4-5`), 則返回空數組
 *
 * @example
 * // 獲取第 10 樓所有的消息頁
 * const chat_messages = getChatMessages(10, { include_swipes: true });
 * const chat_messages = getChatMessages('10', { include_swipes: true });
 *
 * @example
 * // 獲取最新樓層所有的消息頁
 * const chat_message = getChatMessages(-1, { include_swipes: true })[0];  // 或 getChatMessages('{{lastMessageId}}', { include_swipes: true })[0]
 *
 * @example
 * // 獲取所有樓層所有的消息頁
 * const chat_messages = getChatMessages('0-{{lastMessageId}}', { include_swipes: true });
 */
declare function getChatMessages(
  range: string | number,
  { role, hide_state, include_swipes }?: Omit<GetChatMessagesOption, 'include_swipes'> & { include_swipes?: true },
): ChatMessageSwiped[];

/**
 * 獲取聊天消息
 *
 * @param range 要獲取的消息樓層號或樓層範圍, 如 `0`, `'0-{{lastMessageId}}'`, `-1` 等. 負數表示深度, 如 `-1` 表示最新的消息樓層, `-2` 表示倒數第二條消息樓層.
 * @param option 可選選項
 *   - `role:'all'|'system'|'assistant'|'user'`: 按 role 篩選消息; 默認爲 `'all'`
 *   - `hide_state:'all'|'hidden'|'unhidden'`: 按是否被隱藏篩選消息; 默認爲 `'all'`
 *   - `include_swipes:boolean`: 是否包含未被 AI 使用的消息頁信息, 如沒選擇的開局、通過點擊箭頭重 roll 的樓層. 如果不包含則返回類型爲 `ChatMessage`, 否則返回類型爲 `ChatMessageSwiped`; 默認爲 `false`
 *
 * @returns 一個數組, 數組的元素是每樓的消息, 依據 message_id 從低到高排序, 類型爲 `ChatMessage` 或 `ChatMessageSwiped` (取決於 `include_swipes` 的值, 默認爲 `ChatMessage`).
 */
declare function getChatMessages(
  range: string | number,
  { role, hide_state, include_swipes }?: GetChatMessagesOption,
): (ChatMessage | ChatMessageSwiped)[];

type SetChatMessagesOption = {
  /**
   * 是否更新樓層在頁面上的顯示; 默認爲 `'affected'`
   * - `'none'`: 不更新頁面的顯示
   * - `'affected'`: 僅更新被影響樓層的顯示, 更新顯示時會發送 `tavern_events.USER_MESSAGE_RENDERED` 或 `tavern_events.CHARACTER_MESSAGE_RENDERED` 事件
   * - `'all'`: 重新載入整個聊天消息, 將會觸發 `tavern_events.CHAT_CHANGED` 事件
   */
  refresh?: 'none' | 'affected' | 'all';
};

/**
 * 修改聊天消息的數據
 *
 * @param chat_messages 要修改的消息, 必須包含 `message_id` 字段
 * @param option 可選選項
 *   - `refresh:'none'|'affected'|'all'`: 是否更新樓層在頁面上的顯示; 默認爲 `'affected'`
 *
 * @example
 * // 修改第 10 樓被 AI 使用的消息頁的正文
 * await setChatMessages([{message_id: 10, message: '新的消息'}]);
 *
 * @example
 * // 設置開局
 * await setChatMessages([{message_id: 0, swipes: ['開局1', '開局2']}])
 *
 * @example
 * // 切換爲開局 3
 * await setChatMessages([{message_id: 0, swipe_id: 2}]);
 *
 * @example
 * // 重新渲染第 4 樓的前端界面 (利用 `{render: 'affected'}`)
 * await setChatMessages([{message_id: 4}]);
 *
 * @example
 * // 補充倒數第二樓的樓層變量
 * const chat_message = getChatMessages(-2)[0];
 * _.set(chat_message.data, '神樂光好感度', 5);

 *
 * @example
 * // 隱藏所有樓層
 * const last_message_id = getLastMessageId();
 * await setChatMessages(_.range(last_message_id + 1).map(message_id => ({message_id, is_hidden: true})));
 */
declare function setChatMessages(
  chat_messages: Array<{ message_id: number } & (Partial<ChatMessage> | Partial<ChatMessageSwiped>)>,
  { refresh }?: SetChatMessagesOption,
): Promise<void>;

/**
 * 使用酒館當前啓用的預設, 讓 AI 生成一段文本.
 *
 * 該函數在執行過程中將會發送以下事件:
 * - `iframe_events.GENERATION_STARTED`: 生成開始
 * - 若啓用流式傳輸, `iframe_events.STREAM_TOKEN_RECEIVED_FULLY`: 監聽它可以得到流式傳輸的當前完整文本 ("這是", "這是一條", "這是一條流式傳輸")
 * - 若啓用流式傳輸, `iframe_events.STREAM_TOKEN_RECEIVED_INCREMENTALLY`: 監聽它可以得到流式傳輸的當前增量文本 ("這是", "一條", "流式傳輸")
 * - `iframe_events.GENERATION_ENDED`: 生成結束, 監聽它可以得到生成的最終文本 (當然也能通過函數返回值獲得)
 *
 * @param config 提示詞和生成方式設置
 *   - `user_input?:string`: 用戶輸入
 *   - `should_stream?:boolean`: 是否啓用流式傳輸; 默認爲 'false'
 *   - `should_silence?:boolean`: 是否靜默生成; 默認爲 'false'
 *   - `image?:File|string`: 圖片輸入
 *   - `overrides?:Overrides`: 覆蓋選項. 若設置, 則 `overrides` 中給出的字段將會覆蓋對應的提示詞. 如 `overrides.char_description = '覆蓋的角色描述';` 將會覆蓋角色描述
 *   - `injects?:Omit<InjectionPrompt, 'id'>[]`: 要額外注入的提示詞
 *   - `max_chat_history?:'all'|number`: 最多使用多少條聊天曆史
 * @returns 生成的最終文本
 *
 * @example
 * // 請求生成
 * const result = await generate({ user_input: '你好' });
 * console.info('收到回覆: ', result);
 *
 * @example
 * // 圖片輸入
 * const result = await generate({ user_input: '你好', image: 'https://example.com/image.jpg' });
 * console.info('收到回覆: ', result);
 *
 * @example
 * // 注入、覆蓋提示詞
 * const result = await generate({
 *   user_input: '你好',
 *   injects: [{ role: 'system', content: '思維鏈...', position: 'in_chat', depth: 0, should_scan: true, }]
 *   overrides: {
 *     char_personality: '溫柔',
 *     world_info_before: '',
 *     chat_history: {
 *       prompts: [],
 *     }
 *   }
 * });
 * console.info('收到回覆: ', result);
 *
 * @example
 * // 使用自定義API
 * const result = await generate({
 *   user_input: '你好',
 *   custom_api: {
 *     apiurl: 'https://your-proxy-url.com',
 *     key: 'your-api-key',
 *     model: 'gpt-4',
 *     source: 'openai'
 *   }
 * });
 * console.info('收到回覆: ', result);
 *
 * @example
 * // 使用酒館代理預設
 * const result = await generate({
 *   user_input: '你好',
 *   custom_api: {
 *     proxy_preset: 'MyProxy',
 *     model: 'gpt-4',
 *   }
 * });
 * console.info('收到回覆: ', result);
 *
 * @example
 * // 使用當前 ST 源，但切換模型
 * const result = await generate({
 *   user_input: '你好',
 *   custom_api: {
 *     model: 'gpt-4-turbo',
 *   }
 * });
 * console.info('收到回覆: ', result);
 *
 * @example
 * // 流式生成
 *
 * // 需要預先監聽事件來接收流式回覆
 * eventOn(iframe_events.STREAM_TOKEN_RECEIVED_FULLY, text => {
 *   console.info('收到流式回覆: ', text);
 * });
 *
 * // 然後進行生成
 * const result = await generate({ user_input: '你好', should_stream: true });
 * console.info('收到最終回覆: ', result);
 *
 * @example
 * // 使用 tool use / function calling
 * const result = await generate({
 *   user_input: '今天北京天氣怎麼樣？',
 *   tools: [{
 *     type: 'function',
 *     function: {
 *       name: 'get_weather',
 *       description: '獲取指定城市的天氣',
 *       parameters: {
 *         type: 'object',
 *         properties: {
 *           city: { type: 'string', description: '城市名稱' }
 *         },
 *         required: ['city']
 *       }
 *     }
 *   }],
 *   tool_choice: 'auto'
 * });
 *
 * if (typeof result === 'object' && result.tool_calls) {
 *   for (const call of result.tool_calls) {
 *     console.info(`模型調用了工具: ${call.function.name}(${call.function.arguments})`);
 *   }
 * } else {
 *   console.info('收到文本回復: ', result);
 * }
 *
 * @example
 * // 使用 json_schema 強制結構化輸出
 * const result = await generate({
 *   user_input: '描述場景',
 *   json_schema: {
 *     name: 'scene_output',

 *       type: 'object',
 *       properties: {
 *         narrative: { type: 'string' },
 *         mood: { type: 'string', enum: ['happy', 'sad', 'tense'] }
 *       },
 *       required: ['narrative', 'mood']
 *     }
 *   }
 * });

 * console.info(parsed.narrative, parsed.mood);
 */

   * - `iframe_events.STREAM_TOKEN_RECEIVED_FULLY`: 監聽它可以得到流式傳輸的當前完整文本 ("這是", "這是一條", "這是一條流式傳輸")
   * - `iframe_events.STREAM_TOKEN_RECEIVED_INCREMENTALLY`: 監聽它可以得到流式傳輸的當前增量文本 ("這是", "一條", "流式傳輸")
   *
   * @example
   * eventOn(iframe_events.STREAM_TOKEN_RECEIVED_FULLY, text => console.info(text));
   */
  should_stream?: boolean;

  /**

/**
 * JSON Schema 定義，用於強制模型輸出符合指定 schema 的 JSON。
 *
 * @example
 * const result = await generateRaw({
 *   user_input: '描述場景',
 *   json_schema: {
 *     name: 'scene_output',
 *     description: '場景描述和角色狀態',
 *     value: {
 *       type: 'object',
 *       properties: {
 *         narrative: { type: 'string', description: '敘事文本' },
 *         status: { type: 'object', properties: { name: { type: 'string' } } }
 *       },
 *       required: ['narrative', 'status']
 *     }
 *   }
 * });
 * const parsed = JSON.parse(result as string);
 */
type JsonSchema = {
  /** Schema 名稱 */
  name: string;
  /** Schema 描述 */
  description?: string;
  /** JSON Schema 定義 */
  value: Record<string, any>;
  /** 是否嚴格模式（默認 true） */
  strict?: boolean;
};

 * Tool function 定義
 */

  /** 工具函數名稱 */
  name: string;
  /** 工具函數描述 */
  description?: string;
  /** JSON Schema 格式的參數定義 */
  parameters?: Record<string, any>;
};

 * Tool 定義（OpenAI 格式）
 */
type ToolDefinition = {
  type: 'function';
  function: ToolFunction;
};

/**
 * Tool choice 選項
 */
type ToolChoice = 'auto' | 'required' | 'none' | 'any' | { type: 'function'; function: { name: string } };

/**
 * 當模型返回 tool_calls 時的結構化結果。
 *
 * 僅在 `generate` / `generateRaw` 配置中傳入了 `tools` 且模型決定調用工具時返回；
 * 否則函數仍返回普通的 `string`。
 */
type GenerateToolCallResult = {
  /** 模型返回的文本內容（可能爲空字符串） */
  content: string;
  /** 模型請求調用的工具列表 */
  tool_calls: {
    id: string;
    type: 'function';
    function: {
      /** 工具函數名稱 */
      name: string;
      /** JSON 字符串格式的參數 */
      arguments: string;
    };
    /**
     * 加密的 reasoning/thought 簽名（若 provider 返回）。
     *
     * 多輪 tool call 場景下必須把簽名原樣回傳給下一輪請求以維持推理上下文。
     *
     * **Gemini 3 強制要求**：不回傳 thought_signature 會返回 4xx 校驗錯誤
     * （Gemini 2.5 及之前是可選，3.0+ 強制，見官方 thought-signatures 文檔）。
     *
     * **並行 tool call**：Gemini 只會把簽名掛在**第一個** tool_call 上，
     * 後續並行 call 的 `thought_signature` 爲 undefined——這一個簽名代表整批，
     * 回傳時只需要還原到第一個 call 對應的 functionCall part 上。
     *
     * **多輪累積**：必須把**歷史所有輪次**返回過的簽名一起回傳，而不是隻帶最後一次。
     *
     * **繞過校驗**（僅限手動構造 tool call 的情況）：把 thought_signature 設爲
     * `"skip_thought_signature_validator"` 或 `"context_engineering_is_the_way_to_go"`
     * 可以跳過 Gemini 服務端校驗。
     */
    thought_signature?: string;
  }[];
  /**
   * 頂層 reasoning 簽名（非綁定到具體 tool_call 的那一份）。
   *
   * 主要出現在「模型只返回文本、沒有 tool_calls」的場景：Gemini 會把簽名掛在最後一個
   * text part 上（流式終幀裏可能是空字符串 text + 簽名）。OpenRouter 通過
   * `reasoning_details` 暴露非 tool 綁定的 encrypted 段；Claude 通過 thinking 塊的
   * signature 字段提供。
   *
   * 同樣用於多輪場景下把 thinking 上下文回傳給下一輪請求。
   */
  reasoning_signature?: string;
};

type PresetPrompt = {
  /**

   * - 普通提示詞 (`isPresetNormalPrompt`): 預設界面上可以手動添加的提示詞
   * - 系統提示詞 (`isPresetSystemPrompt`): 酒館所設置的系統提示詞, 但其實相比於手動添加的提示詞沒有任何優勢, 分爲 `main`、`nsfw`、`jailbreak`、`enhance_definitions`

   */
  id: TypeFest.LiteralUnion<
    | 'main'
    | 'nsfw'
    | 'jailbreak'
    | 'enhanceDefinitions'
    | 'worldInfoBefore'
    | 'personaDescription'
    | 'charDescription'
    | 'charPersonality'
    | 'scenario'
    | 'worldInfoAfter'
    | 'dialogueExamples'
    | 'chatHistory',
    string
  >;
  name: string;

   * 插入位置, 僅用於普通和佔位符提示詞

   *   - `'in_chat'`: 插入到聊天記錄的對應深度, 需要設置對應的深度 `depth` 和順序 `order`

  position?:
    | {
        type: 'relative';
        depth?: never;
        order?: never;

  content?: string;

type WorldbookEntry = {
  /** uid 是相對於世界書內部的, 不要跨世界書使用 */
  uid: number;
  name: string;
  enabled: boolean;

  /** 激活策略: 條目應該何時激活 */
  strategy: {
    /**
     * 激活策略類型:
     * - `'constant'`: 常量🔵, 俗稱藍燈. 只需要滿足 "啓用"、"激活概率%" 等別的要求即可.
     * - `'selective'`: 可選項🟢, 俗稱綠燈. 除了藍燈條件, 還需要滿足 `keys` 掃描條件
     * - `'vectorized'`: 向量化🔗. 一般不使用
     */
    type: 'constant' | 'selective' | 'vectorized';
    /** 主要關鍵字. 綠燈條目必須在欲掃描文本中掃描到其中任意一個關鍵字才能激活 */
    keys: (string | RegExp)[];
    /**
     * 次要關鍵字. 如果次要關鍵字的 `keys` 數組不爲空, 則條目除了在主要關鍵字中匹配到任意一個關鍵字外, 還需要滿足 `logic`:
     * - `'and_any'`: 次要關鍵字中任意一個關鍵字能在欲掃描文本中匹配到
     * - `'and_all'`: 次要關鍵字中所有關鍵字都能在欲掃描文本中匹配到
     * - `'not_all'`: 次要關鍵字中至少有一個關鍵字沒能在欲掃描文本中匹配到
     * - `'not_any'`: 次要關鍵字中所有關鍵字都沒能欲掃描文本中匹配到
     */
    keys_secondary: { logic: 'and_any' | 'and_all' | 'not_all' | 'not_any'; keys: (string | RegExp)[] };
    /** 掃描深度: 1 爲僅掃描最後一個樓層, 2 爲掃描最後兩個樓層, 以此類推 */
    scan_depth: 'same_as_global' | number;
  };
  /** 插入位置: 如果條目激活應該插入到什麼地方 */
  position: {
    /**
     * 位置類型:
     * - `'before_character_definition'`: 角色定義之前
     * - `'after_character_definition'`: 角色定義之後
     * - `'before_example_messages'`: 示例消息之前
     * - `'after_example_messages'`: 示例消息之後
     * - `'before_author_note'`: 作者註釋之前
     * - `'after_author_note'`: 作者註釋之後
     * - `'at_depth'`: 插入到指定深度
     */
    type:
      | 'before_character_definition'
      | 'after_character_definition'
      | 'before_example_messages'
      | 'after_example_messages'
      | 'before_author_note'
      | 'after_author_note'
      | 'at_depth'
      | 'outlet';
    /** 該條目的消息身份, 僅位置類型爲 `'at_depth'` 時有效 */
    role: 'system' | 'assistant' | 'user';
    /** 該條目要插入的深度, 僅位置類型爲 `'at_depth'` 時有效 */
    depth: number;
    // TODO: 世界書條目的插入: 文檔鏈接
    order: number;
  };

  content: string;

  probability: number;
  /** 遞歸表示某世界書條目被激活後, 該條目的提示詞又激活了其他條目 */
  recursion: {
    /** 禁止其他條目遞歸激活本條目 */
    prevent_incoming: boolean;
    /** 禁止本條目遞歸激活其他條目 */
    prevent_outgoing: boolean;
    /** 延遲到第 n 級遞歸檢查時才能激活本條目 */
    delay_until: null | number;
  };
  effect: {
    /** 黏性: 條目激活後, 在之後 n 條消息內始終激活, 無視激活策略、激活概率% */
    sticky: null | number;
    /** 冷卻: 條目激活後, 在之後 n 條消息內不能再激活 */
    cooldown: null | number;
    /** 延遲: 聊天中至少有 n 樓消息時, 才能激活條目 */
    delay: null | number;
  };

  /** 額外字段, 用於爲世界書條目綁定額外數據 */
  extra?: Record<string, any>;
};

type EventOnReturn = {
  /** 取消監聽 */
  stop: () => void;
};

/**
 * 讓 `listener` 監聽 `event_type`, 當事件發生時自動運行 `listener`;
 * 如果 `listener` 已經在監聽 `event_type`, 則調用本函數不會有任何效果.
 *
 * 當 `eventOn` 所在的前端界面/腳本關閉時, 監聽將會自動卸載.
 *
 * @param event_type 要監聽的事件
 * @param listener 要註冊的函數
 *
 * @example
 * function hello() { alert("hello"); }
 * eventOn(要監聽的事件, hello);
 *
 * @example
 * // 監聽消息接收並彈出 `'hello'`
 * eventOn(tavern_events.MESSAGE_RECEIVED, () => alert('hello'));
 *
 * @example
 * // 消息被修改時監聽是哪一條消息被修改
 * // 酒館事件 tavern_events.MESSAGE_UPDATED 會傳遞被更新的樓層 id
 * eventOn(tavern_events.MESSAGE_UPDATED, message_id => {
 *   alert(`你剛剛更新了第 ${message_id} 條聊天消息對吧😡`);
 * });
 *
 * @returns 後續操作
 *   - `stop`: 取消這個監聽
 */
declare function eventOn<T extends EventType>(event_type: T, listener: ListenerType[T]): EventOnReturn;

/** @deprecated 請使用 `eventOn(getButtonEvent('按鈕名稱'), 函數)` 代替 */
declare function eventOnButton<T extends EventType>(event_type: T, listener: ListenerType[T]): void;

/**
 * 讓 `listener` 監聽 `event_type`, 當事件發生時自動在最後運行 `listener`;
 * 如果 `listener` 已經在監聽 `event_type`, 則調用本函數會將 `listener` 調整爲最後運行.
 *
 * 當 `eventMakeLast` 所在的前端界面/腳本關閉時, 監聽將會自動卸載.
 *
 * @param event_type 要監聽的事件
 * @param listener 要註冊/調整到最後運行的函數
 *
 * @example
 * eventMakeLast(要監聽的事件, 要註冊的函數);
 *
 * @returns 後續操作
 *   - `stop`: 取消這個監聽
 */
declare function eventMakeLast<T extends EventType>(event_type: T, listener: ListenerType[T]): EventOnReturn;

/**
 * 讓 `listener` 監聽 `event_type`, 當事件發生時自動在最先運行 `listener`;
 * 如果 `listener` 已經在監聽 `event_type`, 則調用本函數會將 `listener` 調整爲最先運行.
 *
 * 當 `eventMakeFirst` 所在的前端界面/腳本關閉時, 監聽將會自動卸載.
 *
 * @param event_type 要監聽的事件
 * @param listener 要註冊/調整爲最先運行的函數
 *
 * @example
 * eventMakeFirst(要監聽的事件, 要註冊的函數);
 *
 * @returns 後續操作
 *   - `stop`: 取消這個監聽
 */
declare function eventMakeFirst<T extends EventType>(event_type: T, listener: ListenerType[T]): EventOnReturn;

/**
 * 讓 `listener` 僅監聽下一次 `event_type`, 當該次事件發生時運行 `listener`, 此後取消監聽;
 * 如果 `listener` 已經在監聽 `event_type`, 則調用本函數不會有任何效果.
 *
 * 當 `eventOnce` 所在的前端界面/腳本關閉時, 監聽將會自動卸載.
 *
 * @param event_type 要監聽的事件
 * @param listener 要註冊的函數
 *
 * @example
 * eventOnce(要監聽的事件, 要註冊的函數);
 *
 * @returns 後續操作
 *   - `stop`: 取消這個監聽
 */
declare function eventOnce<T extends EventType>(event_type: T, listener: ListenerType[T]): EventOnReturn;

/**
 * 發送 `event_type` 事件, 同時可以發送一些數據 `data`.
 *
 * 所有正在監聽 `event_type` 消息頻道的都會收到該消息並接收到 `data`.
 *
 * @param event_type 要發送的事件
 * @param data 要隨着事件發送的數據
 *
 * @example
 * // 發送 "角色階段更新完成" 事件, 所有監聽該事件的 `listener` 都會被運行
 * eventEmit("角色階段更新完成");
 *
 * @example
 * // 發送 "存檔" 事件, 並等待所有 `listener` (也許是負責存檔的函數) 執行完畢後才繼續
 * await eventEmit("存檔");
 *
 * @example
 * // 發送時攜帶數據 ["你好", 0]
 * eventEmit("事件", "你好", 0);
 */
declare function eventEmit<T extends EventType>(event_type: T, ...data: Parameters<ListenerType[T]>): Promise<void>;

/**
 * 攜帶 `data` 而發送 `event_type` 事件並等待事件處理結束.
 *
 * @param event_type 要發送的事件
 * @param data 要隨着事件發送的數據
 */
declare function eventEmitAndWait<T extends EventType>(event_type: T, ...data: Parameters<ListenerType[T]>): void;

/**
 * 讓 `listener` 取消對 `event_type` 的監聽; 如果 `listener` 沒有監聽 `event_type`, 則調用本函數不會有任何效果.
 *
 * 前端界面/腳本關閉時會自動卸載所有的事件監聽, 你不必手動調用 `eventRemoveListener` 來移除.
 *
 * @param event_type 要監聽的事件
 * @param listener 要取消註冊的函數
 *
 * @example
 * eventRemoveListener(要監聽的事件, 要取消註冊的函數);
 */
declare function eventRemoveListener<T extends EventType>(event_type: T, listener: ListenerType[T]): void;

/**
 * 取消本 iframe 中對 `event_type` 的所有監聽
 *
 * 前端界面/腳本關閉時會自動卸載所有的事件監聽, 你不必手動調用 `eventClearEvent` 來移除.
 *
 * @param event_type 要取消監聽的事件
 */
declare function eventClearEvent(event_type: EventType): void;

/**
 * 取消本 iframe 中 `listener` 的的所有監聽
 *
 * 前端界面/腳本關閉時會自動卸載所有的事件監聽, 你不必手動調用 `eventClearListener` 來移除.
 *
 * @param listener 要取消註冊的函數
 */
declare function eventClearListener(listener: Function): void;

/**
 * 取消本 iframe 中對所有事件的所有監聽
 *
 * 前端界面/腳本關閉時會自動卸載所有的事件監聽, 你不必手動調用 `eventClearAll` 來移除.
 */
declare function eventClearAll(): void;

declare const Mvu: {
  events: {
    /** 新開聊天對變量初始化時觸發的事件  */
    VARIABLE_INITIALIZED: 'mag_variable_initiailized';

    /** 某輪變量更新開始時觸發的事件 */
    VARIABLE_UPDATE_STARTED: 'mag_variable_update_started';

    /**
     * 某輪變量更新過程中, 對文本成功解析了所有更新命令時觸發的事件
     *
     * @example
     * // 修復 gemini 在中文間加入的 '-'', 如將 '角色.絡-絡' 修復爲 '角色.絡絡'
     * eventOn(Mvu.events.COMMAND_PARSED, commands => {
     *   commands.forEach(command => {
     *     command.args[0] = command.args[0].replace(/-/g, '');
     *   });
     * });
     *
     * @example
     * // 修復繁體字, 如將 '絡絡' 修復爲 '絡絡'
     * eventOn(Mvu.events.COMMAND_PARSED, commands => {
     *   commands.forEach(command => {
     *     command.args[0] = command.args[0].replaceAll('絡絡', '絡絡');
     *   });
     * });
     *
     * @example
     * // 添加新的更新命令
     * eventOn(Mvu.events.COMMAND_PARSED, commands => {
     *   commands.push({
     *     type: 'set',
     *     full_match: `_.set('絡絡.好感度', 5)`,
     *     args: ['絡絡.好感度', 5],
     *     reason: '腳本強行更新',
     *   });
     * });
     */
    COMMAND_PARSED: 'mag_command_parsed';

    /**
     * 某輪變量更新結束時觸發的事件
     *
     * @example
     * // 保持好感度不低於 0
     * eventOn(Mvu.events.VARIABLE_UPDATE_ENDED, variables => {
     *   if (_.get(variables, 'stat_data.角色.絡絡.好感度') < 0) {
     *     _.set(variables, 'stat_data.角色.絡絡.好感度', 0);
     *   }
     * })
     *
     * @example
     * // 保持好感度增幅不超過 3
     * eventOn(Mvu.events.VARIABLE_UPDATE_ENDED, (variables, variables_before_update) => {
     *   const old_value = _.get(variables_before_update, 'stat_data.角色.絡絡.好感度');
     *   const new_value = _.get(variables, 'stat_data.角色.絡絡.好感度');
     *
     *   // 新的好感度必須在 舊好感度-3 和 舊好感度+3 之間
     *   _.set(variables, 'stat_data.角色.絡絡.好感度', _.clamp(new_value, old_value - 3, old_value + 3));
     * });

    VARIABLE_UPDATE_ENDED: 'mag_variable_update_ended';

    /** 即將用更新後的變量更新樓層時觸發的事件  */
    BEFORE_MESSAGE_UPDATE: 'mag_before_message_update';
  };

  /**
   * 獲取變量表, 並將其視爲包含 mvu 數據的 MvuData
   *
   * @param  可選選項
   *   - `type?:'message'|'chat'|'character'|'global'`: 對某一樓層的聊天變量 (`message`)、聊天變量表 (`'chat'`)、角色卡變量 (`'character'`) 或全局變量表 (`'global'`) 進行操作, 默認爲 `'chat'`
   *   - `message_id?:number|'latest'`: 當 `type` 爲 `'message'` 時, 該參數指定要獲取的消息樓層號, 如果爲負數則爲深度索引, 例如 `-1` 表示獲取最新的消息樓層; 默認爲 `'latest'`
   *   - `script_id?:string`: 當 `type` 爲 `'script'` 時, 該參數指定要獲取的腳本 ID; 如果在腳本內調用, 則你可以用 `getScriptId()` 獲取該腳本 ID
   *
   * @returns MvuData 數據表
   *
   * @example
   * // 獲取最新消息樓層的 mvu 數據
   * const message_data = Mvu.getMvuData({ type: 'message', message_id: 'latest' });

   * const message_data = Mvu.getMvuData({ type: 'message', message_id: getCurrentMessageId() });
   */
  getMvuData: (options: VariableOption) => Mvu.MvuData;

  /**
   * 完全替換變量表爲包含 mvu 數據的 `mvu_data` (但如果沒用 `parseMessages` 自行處理變量, 則更建議監聽 mvu 事件來修改 mvu 數據!)
   *
   * @param variables 要用於替換的變量表
   * @param option 可選選項
   *   - `type?:'message'|'chat'|'character'|'global'`: 對某一樓層的聊天變量 (`message`)、聊天變量表 (`'chat'`)、角色卡變量 (`'character'`) 或全局變量表 (`'global'`) 進行操作, 默認爲 `'chat'`
   *   - `message_id?:number|'latest'`: 當 `type` 爲 `'message'` 時, 該參數指定要獲取的消息樓層號, 如果爲負數則爲深度索引, 例如 `-1` 表示獲取最新的消息樓層; 默認爲 `'latest'`
   *   - `script_id?:string`: 當 `type` 爲 `'script'` 時, 該參數指定要獲取的腳本 ID; 如果在腳本內調用, 則你可以用 `getScriptId()` 獲取該腳本 ID

   * _.set(mvu_data, 'stat_data.角色.絡絡.好感度', 30);
   * await Mvu.replaceMvuData(mvu_data, { type: 'message', message_id: 'latest' });
   */
  replaceMvuData: (mvu_data: Mvu.MvuData, options: VariableOption) => Promise<void>;

  /**
   * 解析包含變量更新命令 (`_.set`) 的消息 `message`, 根據它更新 `old_data` 中的 mvu 變量數據
   *
   * @param message 包含 _.set() 命令的消息字符串
   * @param old_data 當前的 MvuData 數據
   *
   * @returns 如果有變量被更新則返回新的 MvuData, 否則返回 `undefined`
   *
   * @example
   * // 修改絡絡好感度爲 30
   * const old_data = Mvu.getMvuData({ type: 'message', message_id: 'latest' });
   * const new_data = await Mvu.parseMessage("_.set('角色.絡絡.好感度', 30); // 強制修改", old_data);
   * await Mvu.replaceMvuData(new_data, { type: 'message', message_id: 'latest' });
   */
  parseMessage: (message: string, old_data: Mvu.MvuData) => Promise<Mvu.MvuData>;

  /**
   * 酒館是否正在進行額外模型解析

  isDuringExtraAnalysis: () => boolean;
